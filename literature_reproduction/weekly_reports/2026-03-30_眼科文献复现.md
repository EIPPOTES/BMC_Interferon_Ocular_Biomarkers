class MultiMAE_Pretraining:
    def __init__(self):
        self.encoder = ViT(patch_size=32, mask_ratio=0.75)
        self.decoder = TransformerDecoder()
    
    def forward(self, oct_image, slo_image, layer_labels):
        # 1. 随机mask 75%的patches
        masked_inputs = self.random_mask([oct_image, slo_image, layer_labels])
        # 2. 多模态编码器处理
        latent = self.encoder(masked_inputs)
        # 3. 多任务解码器重建
        pred = self.decoder(latent, task_heads=['reconstruction', 'segmentation'])
        # 4. 多任务loss
        loss = α*mse(oct_pred, oct) + β*mse(slo_pred, slo) + γ*ce(layer_pred, layer)
        return loss
